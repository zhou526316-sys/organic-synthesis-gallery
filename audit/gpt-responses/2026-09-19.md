function Get-MainFile {
    param([string]$Url,[string]$Target)
    $tmp = "$Target.download"
    Remove-Item -LiteralPath $tmp -Force -ErrorAction SilentlyContinue
    try {
        Invoke-WebRequest -Uri $Url -OutFile $tmp -UseBasicParsing -TimeoutSec 30
    }
    catch {
        & curl.exe -L --proxy "http://127.0.0.1:7899" --retry 3 --connect-timeout 20 -o $tmp $Url
        if ($LASTEXITCODE -ne 0) { throw "Download failed: $Url" }
    }
    if (-not (Test-Path -LiteralPath $tmp)) { throw "Downloaded file missing: $tmp" }
    Move-Item -LiteralPath $tmp -Destination $Target -Force
    Write-Host ("UPDATED  " + $Target) -ForegroundColor Cyan
}

foreach ($item in $files.GetEnumerator()) {
    Get-MainFile -Url $item.Value -Target $item.Key
}

    ".\scripts\SYNC-LOCAL-DIAGNOSTICS.ps1"
)

foreach ($target in $psTargets) {

    $errors = $null
    [void][System.Management.Automation.Language.Parser]::ParseFile(

        [ref]$errors

            Write-Host ("Line {0}, Column {1}: {2}" -f $_.Extent.StartLineNumber,$_.Extent.StartColumnNumber,$_.Message) -ForegroundColor Red
        }
        throw "PowerShell parser check failed: $target"
    }
    Write-Host ("PASS  " + $target) -ForegroundColor Green
}
